Q5(a)(i). Write a function to check whether the input value is an Armstrong number.Q5(a)(i)　＋
Python code
def is_armstrong(num):
    original = num
    total = 0
    digits = len(str(num))
    while num > 0:
        digit = num % 10
        total += digit ** digits
        num //= 10
    if total == original:
        print(original, "is an Armstrong number")
    else:
        print(original, "is not an Armstrong number")

n = int(input("Enter a number:"))
is_armstrong(n)
Q5(a)(ii). Write a function to check whether a number is a palindrome.Q5(a)(ii)　＋
Python code
def is_palindrome(num):
    original = num
    reversed_num = 0
    while num > 0:
        digit = num % 10
        reversed_num = reversed_num * 10 + digit
        num //= 10
    if original == reversed_num:
        print(original, "is a palindrome number")
    else:
        print(original, "is not a palindrome number")

n = int(input("Enter a number:"))
is_palindrome(n)

def factorial(n):
    if n < 0:
        return "Factorial is not defined for negative numbers"
    elif n == 0 or n == 1:
        return 1
    else:
        return n * factorial(n - 1)

num = int(input("Enter a number:"))

Q6(b). Create a file geometry.py to calculate base areas for a square and circle. In another file, write pointyShapeVolume(x, y, squareBase) to calculate the volume of a square pyramid if squareBase is True and a right circular cone if False. Use circleArea and squareArea from the geometry module.Q6(b)　＋
geometry.py
import math

def squareArea(edge):
    return edge * edge

def circleArea(radius):
    return math.pi * radius * radius
main.py
import geometry

def pointyShapeVolume(x, y, squareBase):
    if squareBase:
        base_area = geometry.squareArea(x)
    else:
        base_area = geometry.circleArea(x)

    volume = (1 / 3) * base_area * y
    return volume

def main():
    edge = float(input("Enter square side/edge:"))
    height = float(input("Enter pyramid height:"))
    pyramid_vol = pointyShapeVolume(edge, height, True)
    print(f"Square Pyramid Volume: {pyramid_vol:.2f}")

    radius = float(input("Enter cone radius:"))
    height = float(input("Enter cone height:"))
    cone_vol = pointyShapeVolume(radius, height, False)
    print(f"Cone Volume: {cone_vol:.2f}")

if __name__ == "__main__":
    main()
